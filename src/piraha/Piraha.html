Here we see three examples of how to match a simple pattern and extract data using Piraha. These examples highlight what is similar and what is different among the supported langauges.

JavaC++C
import edu.lsu.cct.piraha.*;

public class example0 {
    public static void main(String[] args) {
        // Allocate a grammar
        Grammar gram = new Grammar();

        // Compile the rules individually
        gram.compile("num","[0-9]+");
        gram.compile("expr","range=\\[{num}:{num}\\]");

        // Attempt a match
        // The matcher holds the parse tree in memory.
        String input = "range=[10:150]";
        Matcher  matcher =
            gram.matcher(input);
        if(matcher.matches()) {
            System.out.println("Match succeeded!");
            int v1,v2;

            Group group = matcher;
            Group ch = group.group(0);
            String s = ch.substring();
            v1 = Integer.parseInt(s);

            ch = group.group(1);
            s = ch.substring();
            v2 = Integer.parseInt(s);

            System.out.println("v1=" + v1 + ",v2=" + v2);
        } else {
            Near near = matcher.near();
            System.err.println("fail at pos=" + near.pos
                + ": found='" + input.charAt(near.pos)
                + "', expected=" + near.expected);
        }
    }
}

#include <iostream>
#include <stdlib.h>
#include "Piraha.hpp"

using namespace piraha;

int main() {
    // Allocate a grammar
    smart_ptr<Grammar> gram = new Grammar();

    // Compile the rules individually
    gram->compile("num","[0-9]+");
    gram->compile("expr","range=\\[{num}:{num}\\]");

    // Attempt a match
    // The matcher holds the parse tree in memory.
    std::string input = "range=[10:150]";
    smart_ptr<Matcher> matcher =
        gram->matcher(input.c_str());
    if(matcher->matches()) {
        std::cout << "Match succeeded!" << std::endl;
        int v1,v2;

        smart_ptr<Group> group = matcher->result();
        smart_ptr<Group> ch = group->group(0);
        std::string s = ch->substring();
        v1 = atoi(s.c_str());

        ch = group->group(1);
        s = ch->substring();
        v2 = atoi(s.c_str());

        std::cout << "v1=" << v1 << ",v2="
            << v2 << std::endl;
    } else {
        smart_ptr<Near> near = matcher->near();
        int pos = near->pos;
        std::string expected = near->expected;
        std::cerr << "fail at pos=" << pos
            << ": found='" << input[pos]
            << "', expected=" << expected << std::endl;
    }
    return 0;
}

#include <stdio.h>
#include <string.h>
#include "Piraha.h"
#include <stdlib.h>
#include <assert.h>

int main() {
    // Allocate a grammar
    pir_grammar_t *gram = pir_alloc_grammar();

    // Compile the rules individually
    pir_compile(gram,"num","[0-9]+");
    pir_compile(gram,"expr","range=\\[{num}:{num}\\]");

    // Attempt a match
    // The matcher holds the parse tree in memory.
    pir_group_t group,ch;
    const char *input = "range=[10:150]";
    const char *pattern_name = 0; // use default
    pir_matcher_t *matcher =
        pir_alloc_matcher(gram,pattern_name,input);
    if(pir_matches(matcher,&group)) {
        printf("Match succeeded!\n");
        char buf[10];
        int v1,v2;

        pir_get_group(&group,0,&ch);
        pir_get_str(&ch,buf);
        v1 = atoi(buf);

        pir_get_group(&group,1,&ch);
        pir_get_str(&ch,buf);
        v2 = atoi(buf);

        printf("v1=%d,v2=%d\n",v1,v2);
    } else {
        char expected[20];
        int pos = pir_error(matcher,expected);
        printf("fail at pos=%d: found=(%c), expected=%s\n",
            pos,input[pos],expected);
    }
    pir_free_matcher(matcher);

    // Free the space used by the Grammar
    pir_free_grammar(gram);
    return 0;
}
